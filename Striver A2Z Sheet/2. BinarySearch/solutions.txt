2004. Ceil The Floor | Medium | ***
TC: O(NlogN + 2*logN)
SC: O(1)
class Solution {
    public int floor(int x, int[] arr){
        int low = 0 , high = arr.length-1;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(arr[mid] <= x){
                low = mid + 1;
            }else{
                high = mid - 1;
            }
        }
        // System.out.println("High: " + high);
        return high >= 0 ? arr[high] : -1;
    }
    
    public int ceil(int x, int[] arr){
        int low = 0 , high = arr.length-1;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(arr[mid] < x){
                low = mid + 1;
            }else{
               high = mid - 1;
            }
        }
        // System.out.println("Low: " + low);
        return low == arr.length ? -1 : arr[low];
    }
    public int[] getFloorAndCeil(int x, int[] arr) {
        Arrays.sort(arr);
        // System.out.println(Arrays.toString(arr));
        return new int[]{floor(x,arr),ceil(x,arr)};
        
    }
}

2006. Find First and Last Position of Element in Sorted Array
TC: O(2*logN)
SC: O(1)
class Solution {
    public int insertPosition(int[] nums, int target){
        int low = 0 , high = nums.length-1;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(nums[mid] < target){
                low = mid + 1;
            }else{
                high = mid - 1;
            }
        }

        return low;
    }
    public int[] searchRange(int[] nums, int target) {
        int startIndex = insertPosition(nums,target);

        if(startIndex >= nums.length || nums[startIndex] != target){
            return new int[]{-1,-1};
        }else{
            return new int[]{startIndex, insertPosition(nums,target+1)-1};
        }
    }
}

            long mid = low + (high-low)/2;

2014.. Find the Nth root of a number using binary search || EASY || *
TC: O(log(N))
SC: O(1);
class Solution {
    public int checkPower(int x , int y, int exp){
        int ans = 1;
        while(y-- > 0){
            ans *= x;
        }
        
        if(ans < exp) return -1;
        else if(ans == exp) return 0;
        else return 1;
    }
    public int nthRoot(int n, int m) {
        if(m == 0 || m == 1) return m;
        int low = 2 , high = m/n;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(checkPower(mid,n,m) <= 0){
                low = mid+1;
            }else{
                high = mid-1;
            }
        }
        
        if(checkPower(high, n, m) == 0) return high;
        return -1;
        
    }
}

        return totalHours;

        int high = Integer.MIN_VALUE;

        for(int pile : piles){
            high = Math.max(high,pile);
        }

            }else{
                high = currRate - 1;
            }
        }

        return low;
    }

2017. Minimum days to make M bouquets || Medium || *
class Solution {
    public int makeBouquet(int currDay, int[] bloomDay, int k){
        int adjFlower = 0;
        int bouqCount = 0;
        for(int bDay : bloomDay){
            if(bDay <= currDay){
                adjFlower++;

                if(adjFlower == k){
                    bouqCount++;
                    adjFlower = 0;
                }
            }else{
                adjFlower = 0;
            }
        }

        if(adjFlower == k){
            bouqCount++;
        }
        return bouqCount;
    }

    public int minDays(int[] bloomDay, int m, int k) {
        if((long)m*k > bloomDay.length) return -1;


        int low = Integer.MAX_VALUE;
        int high = Integer.MIN_VALUE;
        for(int day : bloomDay){
            low = Math.min(low,day);
            high = Math.max(high,day);
        }

        while(low <= high){
            int currDay = low + (high-low)/2;
            int bqPoss = makeBouquet(currDay, bloomDay, k);

            if(bqPoss < m){
                low = currDay + 1;
            }else{
                high = currDay - 1;
            }
        }
        return low;
    }
}

                high = mid - 1;
            }
        }

    }

Aggressive Cows || EASY || *

class Solution {
    public static int aggressiveCows(int[] stalls, int k) {
        Arrays.sort(stalls);
        int n = stalls.length, low = 1, high = stalls[n-1] - stalls[0];
        
        while(low <= high){
            int mid = low + (high - low)/2;
            
            boolean isPossible = calc(stalls, mid, k);
            if(isPossible){
                low = mid + 1;
            }else{
                high = mid - 1;
            }
        }
        
        return high;
    }
    
    public static boolean calc(int[] stalls, int dist, int k){
        k-=1;
        int count = 0, prev = stalls[0];
        
        for(int i = 1 ; i < stalls.length ; i++){
            if(stalls[i] - prev >= dist){
                count++;
                prev = stalls[i];
            }
            
            if(count == k) return true;
        }
        return false;
    }
}

public class Solution {
    public static int calc(ArrayList<Integer> arr, int n, int maxPages){
        long adjPages = 0;
        int m = 0;

        for(int bookPages : arr){
            if(adjPages + bookPages > maxPages){
                m++;
                adjPages = 0;
            }

            adjPages += bookPages;
        }
        return m+1;
    }
    public static int findPages(ArrayList<Integer> arr, int n, int m) {
        if(m > n) return -1;
        int low = 0, high = 0;
        for(int a : arr){
            low = Math.max(low, a);
            high += a;
        }

        while(low <= high){
            int mid = low + (high-low)/2;

            int students = calc(arr, n, mid);

            if(students > m){
                low = mid + 1;
            }else{
                high = mid-1;
            }
        }

        return low;
    }
}

                adjSum = num;

            if(subArrays > k){
                low = mid + 1;
            }else{
                high = mid - 1;
            }
        }
        return low;
    }
}

    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        int n = nums1.length, m = nums2.length;
        int totalEle = n+m;

        int index2 = totalEle / 2;
        int index1 = index2 - 1;

        int ele1 = -1, ele2 = -1;
        int k = 0;
        int i = 0 , j = 0;
        while(i < n && j < m){
            if(nums1[i] <= nums2[j]){
                if(k == index1) ele1 = nums1[i];
                if(k == index2) ele2 = nums1[i];
                i++;
            }else{
                if(k == index1) ele1 = nums2[j];
                if(k == index2) ele2 = nums2[j];
                j++;
            }
            k++;
        }

        while(i < n){
            if(k == index1) ele1 = nums1[i];
            if(k == index2) ele2 = nums1[i]; 
            k++; i++;
        }
        while(j < m){
            if(k == index1) ele1 = nums2[j];
            if(k == index2) ele2 = nums2[j]; 
            k++; j++;
        }

        if(totalEle % 2 == 1) return ele2;

        return ((double)ele2 + (double)ele1)/2;
    }

Row with max 1s || EASY
class Solution {
    public int findFirstOccurence(int arr[][], int row){
        int low = 0 , high = arr[0].length - 1;
        
        while(low <= high){
            int mid = (low + high ) / 2;
            
            if(arr[row][mid] >= 1){
                high = mid - 1;
            }else{
                low = mid + 1;
            }
        }
        
        return low;
    }
    public int rowWithMax1s(int arr[][]) {
        int rowIndex = -1;
        int maxCount = 0;       
        int n = arr.length, m = arr[0].length;
        
        for(int i = 0 ; i < n ; i++){
            int colIndex = findFirstOccurence(arr,i);
            int oneCount = m - colIndex;
            
            if(oneCount > maxCount){
                maxCount = oneCount;
                rowIndex = i;
            }
        }
        
        return rowIndex;
    }
}

    public boolean searchMatrix(int[][] matrix, int target) {

        int low = 0 , high = m*n - 1;

        }
