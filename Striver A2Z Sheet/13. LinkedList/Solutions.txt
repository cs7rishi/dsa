1328 Add two numbers in Linked List || EASY
TC: O(Max(m,n))
SC: O(1)
class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode head = new ListNode(-1);
        ListNode curr = head;
        int carry = 0;

        while(l1 != null || l2 != null || carry != 0){
            int sum = 0;
            sum += l1 == null ? 0 : l1.val;
            sum += l2 == null ? 0 : l2.val;
            sum += carry;
            

            carry = sum / 10;
            sum %= 10;

            curr.next = new ListNode(sum);
            curr = curr.next;

            l1 = l1 != null ? l1.next : null;
            l2 = l2 != null ? l2.next : null;
        }

        return head.next;
    }
}

1329 Segregate odd and even nodes in Linked List || EASY
TC: O(n)
SC: O(1)
class Solution {
    public ListNode oddEvenList(ListNode head) {
        int INF = 1_000_001;
        ListNode headOdd = new ListNode(INF), headEven = new ListNode(INF);
        ListNode tailOdd = headOdd , tailEven = headEven;

        int count = 1;
        while(head != null){
            if(count % 2 == 1){
                tailOdd.next = head;
                tailOdd = head;
            }else{
                tailEven.next = head;
                tailEven = head;
            }

            head = head.next;
            count++;
        }

        tailOdd.next = null;
        tailEven.next = null;

        if(tailOdd.val == INF){
            return headEven.next;
        }else{
            tailOdd.next = headEven.next;
            return headOdd.next;
        }
    }
}

class Solution {
    public Node segregate(Node head) {
        Node head0 = new Node(-1), head1 = new Node(-1) , head2 = new Node(-1);
        Node tail0 = head0, tail1 = head1, tail2 = head2;
        
        // Node curr = head;
        while(head != null){
            if(head.data == 0){
                tail0 = tail0.next = head;
            }else if(head.data == 1){
                tail1 = tail1.next = head;
            }else{
                tail2 = tail2.next = head;
            }
            
            head = head.next;
        }
        
        Node newHead = new Node(-1);
        Node newTail = newHead;
        
        if(tail0.data != -1){
            newTail.next = head0.next;
            newTail = tail0;
        }
        
        if(tail1.data != -1){
            newTail.next = head1.next;
            newTail = tail1;
        }
        
        if(tail2.data != -1){
            newTail.next = head2.next;
            newTail = tail2;
        }
        
        newTail.next = null;
        return newHead.next;
    }
}

1331 Remove Nth node from the back of the LL || EASY || ***
TC: o(n)
SC: O(1)
class Solution {
    public ListNode removeNthFromEnd(ListNode head, int n) {
        ListNode dummy = new ListNode(-1);
        dummy.next = head;
        
        ListNode slow = dummy;
        ListNode fast = dummy;
        
        // 1. Move 'fast' ahead by n + 1 steps. 
        // We do n + 1 so that 'slow' ends up exactly ONE node before the node we want to delete.
        for (int i = 0; i <= n; i++) {
            fast = fast.next;
        }
        
        // 2. Move both pointers at the same speed.
        // When 'fast' hits the end of the list, 'slow' is in the perfect position.
        while (fast != null) {
            slow = slow.next;
            fast = fast.next;
        }
        
        // 3. Remove the target node by skipping it
        slow.next = slow.next.next;
        
        return dummy.next;
    }
}

1336 Check if LL is palindrome or not
TC: O(n)
SC: O(1)
class Solution {
    public boolean isPalindrome(ListNode head) {
        ListNode slow = head;
        ListNode fast = head;
        ListNode prev = null;

        while(fast != null && fast.next != null){
            fast = fast.next.next;

            ListNode nextNode = slow.next;
            slow.next = prev;
            prev = slow;
            slow = nextNode;
        }

        if(fast != null){
            slow = slow.next;
        }

        ListNode rev = prev;
        while(rev != null){
            if(rev.val != slow.val){
                return false;
            }
            rev = rev.next;
            slow = slow.next;
        }

        return true;
    }
}

public class Solution {
    public ListNode detectCycle(ListNode head) {
        if(head == null || head.next == null) return null;
        
        ListNode slow = head;
        ListNode fast = head;

        while(fast != null && fast.next != null){
            slow = slow.next;
            fast = fast.next.next;

            if(slow == fast){
                break;
            }
        }

        if(slow != fast) return null;

        while(head != fast){
            head = head.next;
            fast = fast.next;
        }

        return fast;
    }
}

1340 Length of loop in LL || EASY
TC: O(n)
SC: O(1)
class Solution {
    public int lengthOfLoop(Node head) {
        if(head.next == null) return 0;
        
        Node slow = head, fast = head;
        
        while(fast != null && fast.next != null){
            slow = slow.next;
            fast = fast.next.next;
            
            if(slow == fast){
                break;
            }
        }
        
        if(slow != fast) return 0;
        
        while(head != fast){
            head = head.next;
            fast = fast.next;
            
            if(head == fast) break;
        }
        
        //head becomes cycle starting node
        
        int count = 1;
        fast = fast.next;
        while(fast != head){
            count++;
            
            fast = fast.next;
        }
        return count;
        
    }
}

1341 Reverse LL in group of given size K
TC:
SC:
class Solution {
    // Reverses a detached linked list segment
    public ListNode reverseList(ListNode head) {
        ListNode current = head;
        ListNode prev = null;

        while (current != null) {
            ListNode nextNode = current.next;
            current.next = prev;
            prev = current;
            current = nextNode;
        }

        return prev;
    }

    // Finds the k-th node starting from 'start'
    public ListNode getKthNode(ListNode start, int k) {
        ListNode current = start;
        int count = 1;
        
        while (current != null && count < k) {
            current = current.next;
            count++;
        }
        
        return current;
    }

    public ListNode reverseKGroup(ListNode head, int k) {
        ListNode dummy = new ListNode(-1);
        ListNode prevGroupTail = dummy;
        ListNode groupHead = head;

        while (groupHead != null) {
            ListNode kthNode = getKthNode(groupHead, k);

            // If we have a full group of k nodes
            if (kthNode != null) {
                // 1. Keep track of the start of the next group
                ListNode nextGroupHead = kthNode.next;
                
                // 2. Detach the current group to reverse it safely
                kthNode.next = null;
                reverseList(groupHead); 
                
                // 3. Connect the previous part of the list to the newly reversed group.
                // After reversal, kthNode is the new head of this segment.
                prevGroupTail.next = kthNode;
                
                // 4. Update prevGroupTail for the next iteration.
                // After reversal, the old groupHead is now the tail.
                prevGroupTail = groupHead;
                
                // 5. Move to the next group
                groupHead = nextGroupHead;
                
            } else {
                // If fewer than k nodes remain, just attach them as is and stop
                prevGroupTail.next = groupHead;
                break;
            }
        }

        return dummy.next;
    }
}

1342 Rotate a LL
TC: O(N)
SC: O(1)
class Solution {
    public ListNode rotateRight(ListNode head, int k) {
        if(head == null || head.next == null || k == 0) return head;

        ListNode tail = head;
        int length = 1;
        while(tail.next != null){
            tail = tail.next;
            length++;
        }

        tail.next = head;

        k = k%length;
        int stepToNewTail = length - k;

        ListNode newTail = tail;
        for(int i = 0 ; i < stepToNewTail ; i++){
            newTail = newTail.next;
        }

        ListNode newHead = newTail.next;
        newTail.next = null;

        return newHead;
    }
}

class Solution {
    public Node deleteAllOccurOfX(Node head, int x) {
        if (head == null) return null; // 1. Fix null head
        
        Node dummy = new Node(-1);
        dummy.next = head;
        head.prev = dummy;
        
        Node curr = head, prev = dummy;
        
        while(curr != null){
            if(curr.data == x){
                prev.next = curr.next;
                
                if(curr.next != null){
                    curr.next.prev = prev;
                }
            } else {
                prev = curr;
            }
            
            curr = curr.next;
        }
        
        // 2. Fix potential null pointer if all nodes were deleted
        Node newHead = dummy.next;
        if (newHead != null) {
            newHead.prev = null;
        }
        
        return newHead;
    }
}

class Solution {

        Node curr = headRef;

            if(curr.prev != null && curr.prev.data == curr.data){

1349 Find Pairs with Given Sum in Doubly Linked List
TC: O(n)
SC: O(1)
class Solution {
    public ArrayList<ArrayList<Integer>> givenSumPairs(Node head, int target) {
        Node low = head, high = head;
        ArrayList<ArrayList<Integer>> ans = 
            new ArrayList<>();
        
        while(high.next != null) high = high.next;
        
        while(low.data < high.data){
            int a = low.data, b = high.data;
            if(a+b == target){
                ans.add(new ArrayList<>(List.of(a,b)));
                low = low.next;
                high = high.prev;
            }else if(a + b < target){
                low = low.next;
            }else{
                high = high.prev;
            }
        }
        
        return ans;
        
    }
}

1350. Merge k Sorted Lists
TC:
SC:
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        if (lists == null || lists.length == 0) return null;
        
        PriorityQueue<ListNode> pq = new PriorityQueue<>((a, b) -> a.val - b.val);
        
        // 1. Push the head of each non-empty list into the Priority Queue
        for (ListNode head : lists) {
            if (head != null) {
                pq.add(head);
            }
        }
        
        ListNode dummy = new ListNode(-1);
        ListNode tail = dummy;
        
        // 2. Build the merged list
        while (!pq.isEmpty()) {
            ListNode top = pq.poll();
            
            tail.next = top;
            tail = tail.next;
            
            if (top.next != null) {
                pq.add(top.next);
            }
        }
        
        return dummy.next;
    }
}
